int trie[30 * N][2], cnt[30 * N], sz;
const int mxBit = 30;

int addNode() {
    trie[sz][0] = trie[sz][1] = -1;
    cnt[sz] = 0;
    sz++;
    return sz - 1;
}

void init() {
    sz = 0;
    addNode();
}

// insert or remove
void insert(ll x, int type = 1) {
    int cur = 0;
    cnt[cur] += type;
    for (int i = mxBit; i >= 0; --i) {
        int t = (x >> i) & 1;
        if (trie[cur][t] == -1)
            trie[cur][t] = addNode();
        cur = trie[cur][t];
        cnt[cur] += type;
    }
}

ll maxXor(ll x) {
    int cur = 0;
    if (!cnt[cur])return -1e9;
    for (int i = mxBit; i >= 0; --i) {
        int t = (x >> i) & 1 ^ 1;
        if (trie[cur][t] == -1 || !cnt[trie[cur][t]])t ^= 1;
        cur = trie[cur][t];
        if (t)x ^= 1ll << i;
    }
    return x;
}
