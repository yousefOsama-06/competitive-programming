vector<pair<pair<int, int>, int>> edges;
bool negCycle(int src) {
	for (int i = 1; i <= n; i++) c[i] = inf;
	c[src] = 0;
 
	for (int i = 1; i < n; i++) {
		bool relax = 0;
		for (auto e: edges) {
			if (c[e.first.first] + e.second < c[e.first.second]) {
				c[e.first.second] = c[e.first.first] + e.second;
				relax = 1;
			}
		}
		if (!relax) break;
	}
	for (auto e: edges) {
		if (c[e.first.first] + e.second < c[e.first.second])
			return 1;
	}
	return 0;
}
