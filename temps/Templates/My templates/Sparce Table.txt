int a[N], mn[N][LG], bigPow[N], n;

void con() {
	for (int k = 0; k < LG; k++) {
		for (int i = 0; i + (1 << k) - 1 < n; i++) {
			if (k == 0)
				mn[i][k] = a[i];
			else
				mn[i][k] = min(mn[i][k - 1], mn[i + (1 << (k - 1))][k - 1]);
		}
	}

	bigPow[1] = 0;
	for (int k = 2; k <= n; k++)
		bigPow[k] = bigPow[k / 2] + 1;
}

int query(int l, int r) {
	int len = r - l + 1;
	int k = bigPow[len];
	return min(mn[l][k], mn[r - (1 << k) + 1][k]);
}
