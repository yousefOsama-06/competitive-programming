const int K = 26;

struct Trie {
    struct Node {
        int go[K];
        int freq;

        Node() {
            fill(go, go + K, -1);
            freq = 0;
        }
    };

    vector<Node> aut;

    Trie() {
        aut.emplace_back();
    }

    void insert(string &s) {
        int u = 0; //cur node
        for (auto ch: s) {
            int c = ch - 'a';
            if (aut[u].go[c] == -1) {
                aut[u].go[c] = (int) aut.size();
                aut.emplace_back();
            }
            u = aut[u].go[c];
            aut[u].freq++;
        }
    }

    int query(string &s) {
        int u = 0; //cur node
        for (auto ch: s) {
            int c = ch - 'a';
            if (aut[u].go[c] == -1)
                return 0;
            u = aut[u].go[c];
        }
        return aut[u].freq;
    }

};
