struct SegTree {
    struct node {
        ll mx;

        node() {
            mx = -inf;
        }
    };

    vector<node> tree;
    vector<ll> lazy;
    int n;

    node combine(node a, node b) {
        return a;
    }

    void build(int inputN, vector<int> arr) {
        n = inputN;
        if (n & (n - 1))
            n = 1 << ((int) log2(n) + 1);

        tree.resize(n << 1);
        lazy.resize(n << 1);
        for (int i = 0; i < inputN; i++) {
            tree[i + n].mx = arr[i];
        }
        for (int i = n - 1; i >= 1; i--)
            tree[i] = combine(tree[i << 1], tree[(i << 1) | 1]);
    }

    void propagate(int k, int sl, int sr) {
        if (lazy[k] != 0) {
            tree[k].mx += lazy[k];
            if (sl != sr) {
                lazy[k << 1] += lazy[k];
                lazy[k << 1 | 1] += lazy[k];
            }
        }
        lazy[k] = 0;
    }

    void update(int ql, int qr, int v, int k, int sl, int sr) {
        propagate(k, sl, sr);
        if (ql <= sl && qr >= sr) {
            lazy[k] += v;
            propagate(k, sl, sr);
            return;
        }
        if (qr < sl || sr < ql)
            return;
        int mid = (sl + sr) / 2;
        update(ql, qr, v, k << 1, sl, mid);
        update(ql, qr, v, (k << 1) | 1, mid + 1, sr);
        tree[k] = combine(tree[k << 1], tree[k << 1 | 1]);
    }

    void update(int ql, int qr, int v) {
        update(ql, qr, v, 1, 0, n - 1);
    }

    node query(int ql, int qr, int k, int sl, int sr) {
        propagate(k, sl, sr);
        if (ql <= sl && qr >= sr)
            return tree[k];
        if (qr < sl || sr < ql)
            return node();
        int mid = (sl + sr) / 2;
        return combine(query(ql, qr, k << 1, sl, mid), query(ql, qr, k << 1 | 1, mid + 1, sr));
    }

    node query(int ql, int qr) {
        return query(ql, qr, 1, 0, n - 1);
    }
   int BinWalk(int v, int k, int sl, int sr) {
        if (tree[k].mx < v)
            return -1;
        if (sl == sr)
            return sl;
        int mid = (sl + sr) / 2;
        if (tree[k << 1].mx >= v)
            return BinWalk(v, k << 1, sl, mid);
        else
            return BinWalk(v, k << 1 | 1, mid + 1, sr);
    }
};
