vector<int> adj[N];
int dep[N], anc[N][LG];

void dfs(int u, int p, int d) {
    anc[u][0] = p, dep[u] = d;
    for (int k = 1; k < LG; k++)
        anc[u][k] = anc[anc[u][k - 1]][k - 1];
    for (auto i: adj[u])
        if (i != p)
            dfs(i, u, d + 1);;
}

int binLift(int x, int jump) {
    for (int b = LG - 1; b >= 0; b--)
        if (jump & (1 << b))
            x = anc[x][b];

    return x;
}

int LCA(int a, int b) {
    if (dep[a] > dep[b])
        swap(a, b);
    int diff = dep[b] - dep[a];
    b = binLift(b, diff);
    if (a == b)
        return a;
    for (int bit = LG - 1; bit >= 0; bit--)
        if (anc[a][bit] != anc[b][bit])
            a = anc[a][bit], b = anc[b][bit];

    return anc[a][0];
}
