const int LOG = 64;
struct Basis {
    ll basis[LOG];
    int sz;

    Basis() {
        sz = 0;
        for (int i = LOG - 1; i >= 0; --i) {
            basis[i] = 0;
        }
    }
    void insert(ll x) {
        for (int i = LOG - 1; i >= 0; --i) {
            if ((x & (1ll << i)) == 0) continue;
            if (!basis[i]) {
                basis[i] = x;
                ++sz;
                return;
            }
            x ^= basis[i];
        }
    }
    void make_and(ll x)
    {
        sz = 0;
        vector<ll> newly;
        for (int i = LOG - 1; i >= 0; --i) {
            basis[i] &= x;
            if(basis[i])
                newly.push_back(basis[i]);
            basis[i] = 0;
        }
        for(auto &i:newly) insert(i);
    }
    ll get_max() {
        ll ans = 0;
        for (int i = LOG - 1; i >= 0; --i) {
            if(basis[i] && !(ans & (1ll<<i)))
                ans ^= basis[i];
        }
        return ans;
    }
    ll get_kth(ll k) { // 1-indexed (from smaller)
        if(k < 1 || k > (1ll << sz)) return -1;

        ll mask = 0, tot = 1ll << sz;
        for (int i = LOG - 1; i >= 0; i--) {
            if (basis[i]) {
                int low = tot / 2;
                if ((low < k && (mask & 1ll << i) == 0) ||
                    (low >= k && (mask & 1ll << i))) mask ^= basis[i];
                if (low < k) k -= low;

                tot /= 2;
            }
        }
        return mask;
    }

    ll get_idx(ll x) { // 1-indexed
        ll res=0;
        for(int i = LOG - 1;i>=0;i--)
            if(basis[i])
                res = res * 2 + ((x >> i) & 1);
        return res + 1;
    }

    bool can(ll x)
    {
        for (int i = LOG - 1; i >= 0; --i) {
            if(basis[i] && (x & (1ll<<i)))
                x ^= basis[i];
        }
        return x == 0;
    }
};
